cmake_minimum_required(VERSION 3.16)
project(NocteNet VERSION 0.2.0 LANGUAGES CXX)
add_library(NocteNet src/ArtNetNode.cpp src/PollReply.cpp)
option(NOCTENET_SACN "Build the optional sACN data module" ON)
if(NOCTENET_SACN)
  target_sources(NocteNet PRIVATE src/SacnCodec.cpp src/SacnNode.cpp src/SacnSourceSelector.cpp)
endif()
target_include_directories(NocteNet PUBLIC src)
target_compile_features(NocteNet PUBLIC cxx_std_11)
if(MSVC)
  target_compile_options(NocteNet PRIVATE /W4)
else()
  target_compile_options(NocteNet PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(NOCTENET_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(NOCTENET_SANITIZERS AND NOT MSVC)
  target_compile_options(NocteNet PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(NocteNet PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(noctenet_tests tests/test_artnet.cpp)
  target_link_libraries(noctenet_tests PRIVATE NocteNet)
  foreach(case dmx diagnostics oversize poll rebind transmit rdm-tod reply-profiles malformed-sweep)
    add_test(NAME ${case} COMMAND noctenet_tests ${case})
  endforeach()
  if(NOCTENET_SACN)
    add_executable(noctenet_sacn_tests tests/test_sacn.cpp)
    target_link_libraries(noctenet_sacn_tests PRIVATE NocteNet)
    foreach(case wire parser sources timeout network receive transmit legacy sweep)
      add_test(NAME sacn-${case} COMMAND noctenet_sacn_tests ${case})
    endforeach()
  endif()
endif()
